idf_component_register(SRCS "radio_init.c" INCLUDE_DIRS "." REQUIRES esp_system esp_app_format esp_wifi bt nvs_flash esp_netif esp_event)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/radio_init.c" REFERENCE_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/radio_init.c")
if(NOT DEFINED REFERENCE_RADIO_MODE)
    set(REFERENCE_RADIO_MODE "wifi")
endif()
if(REFERENCE_RADIO_MODE STREQUAL "wifi")
    set(REFERENCE_WIFI 1)
    set(REFERENCE_BLE 0)
elseif(REFERENCE_RADIO_MODE STREQUAL "ble")
    set(REFERENCE_WIFI 0)
    set(REFERENCE_BLE 1)
elseif(REFERENCE_RADIO_MODE STREQUAL "both")
    set(REFERENCE_WIFI 1)
    set(REFERENCE_BLE 1)
else()
    message(FATAL_ERROR "REFERENCE_RADIO_MODE must be wifi, ble or both")
endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE REFERENCE_SOURCE_SHA256="${REFERENCE_SOURCE_SHA256}"
    REFERENCE_RADIO_MODE="${REFERENCE_RADIO_MODE}" REFERENCE_WIFI=${REFERENCE_WIFI} REFERENCE_BLE=${REFERENCE_BLE})
