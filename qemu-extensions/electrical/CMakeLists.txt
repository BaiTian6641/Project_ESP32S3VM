cmake_minimum_required(VERSION 3.21)
project(esp32s3vm_electrical_dc LANGUAGES C)
enable_testing()
add_library(esp32s3vm_net_dc STATIC net-dc.c)
target_compile_features(esp32s3vm_net_dc PUBLIC c_std_11)
target_include_directories(esp32s3vm_net_dc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(esp32s3vm_net_dc PRIVATE -Wall -Wextra -Werror -pedantic -fno-fast-math)
    target_link_libraries(esp32s3vm_net_dc PUBLIC m)
endif()
add_executable(net_dc_test tests/net-dc-test.c)
target_link_libraries(net_dc_test PRIVATE esp32s3vm_net_dc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(net_dc_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME electrical_dc COMMAND net_dc_test)
set_tests_properties(electrical_dc PROPERTIES TIMEOUT 10)

# H-NET-01: strict v3 graph -> primitive/terminal adapter.
add_library(esp32s3vm_net_adapter STATIC net-adapter.c)
target_compile_features(esp32s3vm_net_adapter PUBLIC c_std_11)
target_link_libraries(esp32s3vm_net_adapter PUBLIC esp32s3vm_net_dc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(esp32s3vm_net_adapter PRIVATE -Wall -Wextra -Werror -pedantic -fno-fast-math)
endif()
add_executable(net_adapter_test tests/net-adapter-test.c)
target_link_libraries(net_adapter_test PRIVATE esp32s3vm_net_adapter)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(net_adapter_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME electrical_adapter COMMAND net_adapter_test)
set_tests_properties(electrical_adapter PROPERTIES TIMEOUT 30)

# ANALOG-02 / H-ANALOG-01: bounded linear RC transients with persistent
# capacitor state, deadline-exact steps and threshold crossing refinement.
add_library(esp32s3vm_net_rc STATIC net-rc.c)
target_compile_features(esp32s3vm_net_rc PUBLIC c_std_11)
target_include_directories(esp32s3vm_net_rc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(esp32s3vm_net_rc PRIVATE -Wall -Wextra -Werror -pedantic -fno-fast-math)
    target_link_libraries(esp32s3vm_net_rc PUBLIC m)
endif()
add_executable(net_rc_test tests/net-rc-test.c)
target_link_libraries(net_rc_test PRIVATE esp32s3vm_net_rc)
# The suite carries an independent fixed-step reference integration; keep it
# fast without touching floating-point semantics (no fast-math anywhere).
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(net_rc_test PRIVATE -Wall -Wextra -Werror -pedantic -O2 -fno-fast-math)
endif()
add_test(NAME electrical_rc COMMAND net_rc_test)
set_tests_properties(electrical_rc PROPERTIES TIMEOUT 300)

# ASan+UBSan variants of both kernels and their suites. These are separate
# instrumented builds: the release objects stay uninstrumented.
option(ELECTRICAL_SANITIZERS "Build ASan+UBSan test variants" ON)
set(ESP32S3VM_SANITIZER_FLAGS -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
if(ELECTRICAL_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    include(CheckCSourceCompiles)
    set(CMAKE_REQUIRED_LINK_OPTIONS ${ESP32S3VM_SANITIZER_FLAGS})
    check_c_source_compiles("int main(void){return 0;}" ESP32S3VM_HAVE_SANITIZERS)
    unset(CMAKE_REQUIRED_LINK_OPTIONS)
    if(ESP32S3VM_HAVE_SANITIZERS)
        add_library(esp32s3vm_net_dc_sanitized STATIC net-dc.c)
        target_compile_features(esp32s3vm_net_dc_sanitized PUBLIC c_std_11)
        target_include_directories(esp32s3vm_net_dc_sanitized PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_options(esp32s3vm_net_dc_sanitized PRIVATE ${ESP32S3VM_SANITIZER_FLAGS} -Wall -Wextra -Werror -pedantic -fno-fast-math)
        target_link_options(esp32s3vm_net_dc_sanitized PUBLIC ${ESP32S3VM_SANITIZER_FLAGS})
        target_link_libraries(esp32s3vm_net_dc_sanitized PUBLIC m)
        add_executable(net_dc_sanitized tests/net-dc-test.c)
        target_link_libraries(net_dc_sanitized PRIVATE esp32s3vm_net_dc_sanitized)
        target_compile_options(net_dc_sanitized PRIVATE -Wall -Wextra -Werror -pedantic)
        add_test(NAME electrical_dc_sanitized COMMAND net_dc_sanitized)
        set_tests_properties(electrical_dc_sanitized PROPERTIES TIMEOUT 120)

        add_library(esp32s3vm_net_rc_sanitized STATIC net-rc.c)
        target_compile_features(esp32s3vm_net_rc_sanitized PUBLIC c_std_11)
        target_include_directories(esp32s3vm_net_rc_sanitized PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_options(esp32s3vm_net_rc_sanitized PRIVATE ${ESP32S3VM_SANITIZER_FLAGS} -Wall -Wextra -Werror -pedantic -fno-fast-math)
        target_link_options(esp32s3vm_net_rc_sanitized PUBLIC ${ESP32S3VM_SANITIZER_FLAGS})
        target_link_libraries(esp32s3vm_net_rc_sanitized PUBLIC m)
        add_executable(net_rc_sanitized tests/net-rc-test.c)
        target_link_libraries(net_rc_sanitized PRIVATE esp32s3vm_net_rc_sanitized)
        target_compile_options(net_rc_sanitized PRIVATE -Wall -Wextra -Werror -pedantic -O2)
        add_test(NAME electrical_rc_sanitized COMMAND net_rc_sanitized)
        set_tests_properties(electrical_rc_sanitized PROPERTIES TIMEOUT 300)

        add_library(esp32s3vm_net_adapter_sanitized STATIC net-adapter.c)
        target_compile_features(esp32s3vm_net_adapter_sanitized PUBLIC c_std_11)
        target_compile_options(esp32s3vm_net_adapter_sanitized PRIVATE ${ESP32S3VM_SANITIZER_FLAGS} -Wall -Wextra -Werror -pedantic -fno-fast-math)
        target_link_options(esp32s3vm_net_adapter_sanitized PUBLIC ${ESP32S3VM_SANITIZER_FLAGS})
        target_link_libraries(esp32s3vm_net_adapter_sanitized PUBLIC esp32s3vm_net_dc_sanitized)
        add_executable(net_adapter_sanitized tests/net-adapter-test.c)
        target_link_libraries(net_adapter_sanitized PRIVATE esp32s3vm_net_adapter_sanitized)
        target_compile_options(net_adapter_sanitized PRIVATE -Wall -Wextra -Werror -pedantic)
        add_test(NAME electrical_adapter_sanitized COMMAND net_adapter_sanitized)
        set_tests_properties(electrical_adapter_sanitized PROPERTIES TIMEOUT 120)
    else()
        message(STATUS "Sanitizer runtime unavailable; ELECTRICAL_SANITIZERS targets skipped")
    endif()
endif()
